def mat(name, color, metallic=0.0, rough=0.45, emission=None):
    m = bpy.data.materials.new(name)
    m.diffuse_color = (*color, 1)
    m.use_nodes = True
    bs = m.node_tree.nodes.get("Principled BSDF")
    bs.inputs["Base Color"].default_value = (*color, 1)
    bs.inputs["Metallic"].default_value = metallic
    bs.inputs["Roughness"].default_value = rough
    if emission:
        bs.inputs["Emission Color"].default_value = (*emission, 1)
        bs.inputs["Emission Strength"].default_value = 2.0
    return m

# ---------- HELPERS ----------
def cube(name, loc, scale, material, bevel=0.0):
    bpy.ops.mesh.primitive_cube_add(location=loc)
    o = bpy.context.object
    o.name = name
    o.dimensions = scale
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    if material: o.data.materials.append(material)
    if bevel:
        mod = o.modifiers.new("Edge bevel", 'BEVEL')
        mod.width = bevel
        mod.segments = 3
    return o

def cyl(name, loc, radius, depth, material, vertices=32, rot=(0,0,0)):
    bpy.ops.mesh.primitive_cylinder_add(vertices=vertices, radius=radius, depth=depth, location=loc, rotation=rot)
    o=bpy.context.object; o.name=name
    if material: o.data.materials.append(material)
    return o

def uv_sphere(name, loc, scale, material):
    bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, location=loc)
    o=bpy.context.object; o.name=name; o.scale=scale
    if material: o.data.materials.append(material)
    return o

def rod_between(name, a, b, radius, material):
    a,b=Vector(a),Vector(b)
    d=b-a
    mid=(a+b)/2
    o=cyl(name, mid, radius, d.length, material, 24)
    o.rotation_mode='QUATERNION'
    o.rotation_quaternion=d.to_track_quat('Z','Y')
    return o

def text_obj(body, loc, size=6, material=TEXTMAT, rot=(math.radians(72),0,0), align='CENTER'):
    bpy.ops.object.text_add(location=loc, rotation=rot)
    t=bpy.context.object
    t.data.body=body
    t.data.align_x=align
    t.data.size=size
    t.data.extrude=0.15
    t.data.bevel_depth=0.03
    t.data.materials.append(material)
    return t

def empty(name, loc):
    o=bpy.data.objects.new(name, None)
    bpy.context.collection.objects.link(o)
    o.location=loc
    return o

# ---------- SOLAR PANEL DEPLOYMENT MECHANISM ----------
# True unfold mechanism:
# Each solar panel is mounted to a vertical hinge on the +X / -X side.
# The panel starts stowed against the CubeSat wall and unfolds 90 degrees
# around the vertical Z axis.  A pivot Empty is used so the panel and ALL
# solar cells rotate together around the actual hinge line.

HINGE_Z = 0
HINGE_X = 50

# Left/right vertical hinge barrels
for side in (-1, 1):
    x = side * HINGE_X
    for z in (-34, 0, 34):
        cyl(f"Hinge barrel {side} {z}", (x,0,z), 4, 22, SILVER, 32, rot=(0,0,0))

# Continuous vertical deployment shafts
for side in (-1, 1):
    cyl(f"Vertical deployment shaft {side}", (side*HINGE_X,0,0),
        2.2, 82, WIRE, 32, rot=(0,0,0))

def make_solar_panel(name, side):
    # Pivot is exactly on the spacecraft side wall.
    pivot = empty(f"{name} HINGE PIVOT", (side*HINGE_X, 0, HINGE_Z))

    # In the stowed position the panel lies vertically against the side.
    # X thickness, Y width, Z height.
    panel_center = (side*(HINGE_X + 32), 0, 0)
    panel = cube(name, panel_center, (3,88,64), DARK, 1.5)
    panel.parent = pivot

    # Solar cells are children of the same pivot, so they unfold with the panel.
    # Put them slightly outside the panel surface.
    for row in range(4):
        for col in range(3):
            y = -31 + row*21
            z = -24 + col*16
            cell = cube(
                f"{name} cell R{row+1}C{col+1}",
                (side*(HINGE_X + 34), y, z),
                (1,17,14),
                SOLAR, 0.4
            )
            cell.parent = pivot

    # Cell grid / bus strips
    for z in (-32, -16, 0, 16, 32):
        strip = cube(
            f"{name} cell bus {z}",
            (side*(HINGE_X + 34.8), 0, z),
            (1,80,0.8),
            CELLGRID, 0.15
        )
        strip.parent = pivot

    # Hinge-side mounting brackets
    for z in (-25, 0, 25):
        bracket = cube(
            f"{name} hinge bracket {z}",
            (side*(HINGE_X + 3), 0, z),
            (7,18,5),
            SILVER, 0.8
        )
        bracket.parent = pivot

    # Store metadata on the pivot for easy animation/debugging.
    pivot["deployment_angle_degrees"] = 90.0
    pivot["deployment_axis"] = "Z"
    pivot["stowed_state"] = "panel against CubeSat side"
    pivot["deployed_state"] = "panel unfolded 90 degrees"

    return pivot

# ---------- ANIMATION: SMA-DRIVEN 90° PANEL UNFOLD ----------
# Frame 1   : fully stowed against the CubeSat
# Frame 20  : SMA heating / trigger
# Frame 45  : panel halfway open
# Frame 70  : fully unfolded at 90°
#
# The pivot Empty is animated, not the panel itself. This makes the panel,
# cells, brackets, and all attached geometry rotate around the real hinge.
for pivot, side in [(left_panel,-1),(right_panel,1)]:
    pivot.rotation_mode = 'XYZ'

    # Stowed
    pivot.rotation_euler[2] = 0
    pivot.keyframe_insert(data_path="rotation_euler", frame=1, index=2)

    # Hold while SMA heats
    pivot.rotation_euler[2] = 0
    pivot.keyframe_insert(data_path="rotation_euler", frame=20, index=2)

    # Smooth 90-degree unfold
    pivot.rotation_euler[2] = math.radians(side * 90)
    pivot.keyframe_insert(data_path="rotation_euler", frame=70, index=2)

    # Make the deployment motion easy to see and physically intentional.
    if pivot.animation_data and pivot.animation_data.action:
        for fc in pivot.animation_data.action.fcurves:
            for kp in fc.keyframe_points:
                kp.interpolation = 'BEZIER'

# ---------- ORGANIZE COLLECTIONS ----------
def move_to_collection(obj, collection):
    for c in list(obj.users_collection):
        c.objects.unlink(obj)
    collection.objects.link(obj)

# Create collections

cols={}
for n in names:
    c=bpy.data.collections.get(n) or bpy.data.collections.new(n)
    if c.name not in scene.collection.children:
        try: scene.collection.children.link(c)
        except: pass
    cols[n]=c

# Move based on object name
for o in list(scene.objects):
    if o.name.startswith(("TEMP","Humidity","Ambient","Barometric","DS18","SMA temperature","Hall","IMU")) or "sensor" in o.name.lower():
        target=cols["03_Sensors"]
    elif "SMA" in o.name or "linkage" in o.name.lower() or "Hinge" in o.name or "Deployment" in o.name:
        target=cols["04_SMA_Deployment"]
    elif "SOLAR" in o.name or "solar" in o.name.lower():
        target=cols["05_Solar_Panels"]
    elif "label" in o.name.lower() or o.type=='FONT':
        target=cols["06_Labels"]
    else:
        target=cols["01_CubeSat_Body"]
    try: move_to_collection(o,target)
    except: pass
